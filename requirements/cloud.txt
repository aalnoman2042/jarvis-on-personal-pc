# VONDO cloud core — what gets deployed to Fly.io.
#
# Deliberately small and platform-neutral: this list must install on Linux with
# no microphone, no screen and no Windows. Anything that touches the desktop
# belongs in agent.txt, not here.

python-dotenv>=1.0.0       # settings from .env
psutil>=5.9.0              # the server's own health, not the PC's

# --- the database ---
# Only needed in the cloud, where there is no disk to keep a SQLite file on.
# Turso speaks SQLite, so every query written for the local file works unchanged
# against it. On Rohan's PC this is unused: db.py falls back to plain sqlite3.
libsql-client>=0.3.1

# --- meaning-based recall ---
# The one honest way to compare a query against thousands of stored vectors in
# the few milliseconds a turn can spare. Hand-rolling the float maths to save a
# dependency would cost roughly a second per question. Platform-neutral, no
# desktop anything, so it does not break the rule this file exists for.
numpy>=1.26.0

# --- reading documents ---
# Pure Python, 3.5MB, no compiled extensions and no system libraries, so it
# installs on the same Linux box as everything else. Parsing PDFs in the browser
# instead was the alternative; doing it here keeps the chunking testable.
pypdf>=5.0.0

# --- the API ---
fastapi>=0.115.0
uvicorn[standard]>=0.30.0  # includes the websocket implementation
# Pinned rather than left to arrive transitively. uvicorn's websocket behaviour
# — including whether its ping settings do anything — depends on which
# implementation pip happened to resolve that day, and "it worked last deploy"
# is not a property you want your PC link to have.
websockets>=13.0
pydantic>=2.0.0
python-multipart>=0.0.9    # FastAPI needs this to accept an uploaded audio clip

# Web Push, which is the only way to reach a CLOSED progressive web app: it has
# no process, no alarms and no way to wake itself. Needs no Firebase and no
# account — the VAPID key pair is generated by the server on first use and the
# browser hands out its own endpoint.
pywebpush>=2.0.0

# --- brains (free tiers first) ---
groq>=0.11.0               # primary
# One SDK, any number of providers. Cerebras, OpenRouter, Together, Mistral,
# Nebius and most others speak the OpenAI protocol, so a second and third free
# tier is a line of configuration rather than a new brain — which is the whole
# answer to "what answers when the free tiers are spent".
openai>=1.40.0
google-genai>=1.0.0        # backup
